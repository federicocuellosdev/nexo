# Nexo 3.0 — Portal privado — Plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construir la sección privada de la maqueta Nexo 3.0 — callback del magic link, sesión mock, bienvenida, pago simulado, wizard de perfil Previnca y las cinco secciones del portal (Inicio, Planes, Coberturas, Movimientos, Cuenta) más la pantalla de control `/demo`.

**Architecture:** SPA Vite + React 19 con React Router 7 en modo librería, estilo MercadoPago: `BottomNav` en mobile y `Sidebar` desde `lg`. **Tech Stack:** Vite 7.3.6 · React 19.2.8 · TypeScript 5.9.3 (strict) · Tailwind v4.3.3 vía `@tailwindcss/vite` con `@theme` · React Router 7.18.3 · `@radix-ui/react-dialog` 1.1.23 · `@radix-ui/react-tabs` 1.1.21 · lucide-react 1.42.0 · Vitest 4.1.11 + Testing Library 16.3.3 + jsdom 27.4.0. Sin Framer Motion. Playwright vive solo en el repo landing.

```ts
const LANDING_POR_DEFECTO = 'http://localhost:5173';

/**
 * Origen del sitio público. Se lee en cada llamada (no como constante de
 * módulo) para que los tests puedan alternarlo con `vi.stubEnv`.
 */
export function landingUrl(): string {
  const desdeEntorno = import.meta.env.VITE_LANDING_URL;
  const base =
    typeof desdeEntorno === 'string' && desdeEntorno.trim().length > 0
      ? desdeEntorno.trim()
      : LANDING_POR_DEFECTO;
  return base.replace(/\/+$/, '');
}
```

```ts
  try {
    const crudo = localStorage.getItem(CLAVE_SESION);
    if (crudo === null) return VACIA;
    const parseado: unknown = JSON.parse(crudo);
    if (typeof parseado !== 'object' || parseado === null) return VACIA;
    const guardada = parseado as Partial<Session>;
    return {
      ...VACIA,
      ...guardada,
      // Normalizar: una sesión vieja o a medio escribir no tiene por qué dejar
      // `ordenes` undefined y hacer explotar cada `.map` del portal.

  } catch {
    return VACIA;
  }
}

  try {
    localStorage.setItem(CLAVE_SESION, JSON.stringify(sesion));
  } catch {
    // Storage bloqueado (modo privado): la maqueta sigue andando en memoria.
  }

  for (const oyente of oyentes) oyente();

  return () => {

  escribirEnStorage(estado);

  }
```

```ts
// Importación SOLO de tipos: se borra al compilar, así que `lib/` sigue siendo
// una hoja y no arrastra `mock/` al bundle de nadie.
import type { Session } from '@/mock/session';

/**
 * Los cuatro estados del spec §5. Se derivan de la sesión en vez de guardarse
 * como campo: un campo `estado` se desincroniza en cuanto alguien agrega una
 * orden y se olvida de actualizarlo.
 */
export type EstadoCuenta = 'sinCuenta' | 'conCuentaSinPago' | 'pagadoSinPerfil' | 'activo';

export function derivarEstado(sesion: Session): EstadoCuenta {
  if (sesion.usuario === undefined) return 'sinCuenta';

  const tienePago = sesion.ordenes.some((orden) => orden.estado === 'pagada');
  if (!tienePago) return 'conCuentaSinPago';

  if (sesion.perfil === undefined) return 'pagadoSinPerfil';

  return 'activo';
}

export function coberturaActiva(sesion: Session): boolean {
  return derivarEstado(sesion) === 'activo';
}
```

```ts
export interface UsuarioCallback {
  nombre: string;
  apellido: string;
  email: string;
  telefono: string;
}

export interface CallbackAlta {
  intent: 'alta';
  tipo: 'plan' | 'producto';
  id: string;
  usuario: UsuarioCallback;
}

export interface CallbackLogin {
  intent: 'login';
  email: string;
}

export type Callback = CallbackAlta | CallbackLogin;

const TOKEN_DEMO = 'demo';

/**
 * Parseo del enlace mágico. Devuelve `null` ante CUALQUIER inconsistencia en
 * vez de completar con valores por defecto: sembrar una sesión a medias con
 * datos inventados sería peor que mandar al usuario a `/ingresar`.
 *
 * Los nombres de parámetro son el contrato con `lib/handoff.ts` de la landing
 * (spec §4). El teléfono viaja como `tel`, no como `telefono`.
 */
export function parsearCallback(search: string): Callback | null {
  const parametros = new URLSearchParams(search);

  if (parametros.get('token') !== TOKEN_DEMO) return null;

  const intent = parametros.get('intent');
  const email = (parametros.get('email') ?? '').trim();

  if (intent === 'login') {
    if (email === '') return null;
    return { intent: 'login', email };
  }

  if (intent !== 'alta') return null;

  const tipo = parametros.get('tipo');
  const id = (parametros.get('id') ?? '').trim();

  if (tipo !== 'plan' && tipo !== 'producto') return null;
  if (id === '' || email === '') return null;

  return {
    intent: 'alta',
    tipo,
    id,
    usuario: {
      nombre: (parametros.get('nombre') ?? '').trim(),
      apellido: (parametros.get('apellido') ?? '').trim(),
      email,
      telefono: (parametros.get('tel') ?? '').trim(),
    },
  };
}

/** Enlace de ingreso desde `/ingresar`: mismo origen, así que es una ruta relativa. */
export function construirCallbackLogin(email: string): string {
  const parametros = new URLSearchParams();
  parametros.set('token', TOKEN_DEMO);
  parametros.set('intent', 'login');
  parametros.set('email', email.trim());
  return `/auth/callback?${parametros.toString()}`;
}
```

```tsx
import { useEffect } from 'react';
import { useLocation, useNavigate } from 'react-router';
import { parsearCallback } from '@/lib/callback';
import { getSession, setSession } from '@/mock/session';
import { Skeleton } from '@/ui/Skeleton';

/** Espera simulada de validación del token (spec §6). */
const MS_VALIDACION = 600;

export function AuthCallback() {
  const { search } = useLocation();
  const navegar = useNavigate();

  useEffect(() => {
    const temporizador = window.setTimeout(() => {
      const datos = parsearCallback(search);

      if (datos === null) {
        void navegar('/ingresar', { replace: true });
        return;
      }

      if (datos.intent === 'alta') {
        setSession({
          usuario: datos.usuario,
          intent: { tipo: datos.tipo, id: datos.id },
        });
        void navegar('/bienvenida', { replace: true });
        return;
      }

      // Login: si ya había un usuario con ese email, se conserva su nombre real
      // en vez de pisarlo con campos vacíos (el enlace de login solo trae email).
      const previo = getSession().usuario;
      const usuario =
        previo !== undefined && previo.email === datos.email
          ? previo
          : { nombre: '', apellido: '', email: datos.email, telefono: '' };

      setSession({ usuario });
      void navegar('/inicio', { replace: true });
    }, MS_VALIDACION);

    return () => {
      window.clearTimeout(temporizador);
    };
  }, [search, navegar]);

  return (
    <div className="mx-auto flex min-h-dvh max-w-[420px] flex-col justify-center px-4">
      <p role="status" className="text-center text-base text-ink-700">
        Validando tu enlace…
      </p>
      <div className="mt-6 flex flex-col gap-3">
        <Skeleton className="h-11 w-full" />
        <Skeleton className="h-11 w-3/4" />
        <Skeleton className="h-11 w-1/2" />
      </div>
    </div>
  );
}
```

```tsx
import { construirCallbackLogin } from '@/lib/callback';
import { RevisaTuMail } from '@/features/ingreso/RevisaTuMail';
import { Field } from '@/ui/Field';

const EMAIL = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/;

function validarEmail(email: string): string | undefined {
  const valor = email.trim();
  if (valor === '') return 'Ingresá tu email';
  if (!EMAIL.test(valor)) return 'Revisá el email: falta el @ o el dominio';
  return undefined;
}

export function Ingresar() {
  const [email, setEmail] = useState('');
  const [error, setError] = useState<string | undefined>(undefined);
  const [enviado, setEnviado] = useState(false);

  function enviar(evento: FormEvent<HTMLFormElement>) {
    evento.preventDefault();
    const encontrado = validarEmail(email);
    if (encontrado !== undefined) {
      setError(encontrado);
      return;
    }
    setError(undefined);
    setEnviado(true);
  }

  return (
    <div className="mx-auto flex min-h-dvh max-w-[440px] flex-col justify-center px-4 py-12">
      {enviado ? (
        <RevisaTuMail email={email.trim()} hrefEnlace={construirCallbackLogin(email)} />
      ) : (
        <>
          <h1 className="text-2xl text-ink-950">Ingresá a tu portal</h1>
          <p className="mt-2 text-sm text-ink-600">
            Te mandamos un enlace por mail. No hace falta contraseña.
          </p>

          <form onSubmit={enviar} noValidate className="mt-6 flex flex-col gap-4">
            <Field label="Email" htmlFor="ingresar-email" error={error} requerido>
              <Input
                id="ingresar-email"
                name="email"
                type="email"
                inputMode="email"
                autoComplete="email"
```

```ts
import type { PerfilPrevinca, Sexo, TipoContratacion, TipoDocumento } from '@/mock/session';

export type PasoPerfil = 1 | 2 | 3 | 4;

/**
 * Borrador plano (sin `domicilio` anidado) porque un formulario controlado es
 * mucho más simple sobre campos planos. `aPerfil` lo convierte al modelo del
 * spec §5 recién al guardar.
 *
 * Los `| ''` representan "todavía no elegido" en los selects.
 */
export interface BorradorPerfil {
  tipoDocumento: TipoDocumento;
  numeroDocumento: string;
  sexo: Sexo | '';
  fechaNacimiento: string;
  calle: string;
  numero: string;
  piso: string;
  localidad: string;
  provincia: string;
  cp: string;
  tieneObraSocial: 'si' | 'no' | '';
  obraSocial: string;
  tipoContratacion: TipoContratacion | '';
}

export type ErroresPerfil = Partial<Record<keyof BorradorPerfil, string>>;

export const BORRADOR_INICIAL: BorradorPerfil = {
  tipoDocumento: 'DNI',
  numeroDocumento: '',
  sexo: '',
  fechaNacimiento: '',
  calle: '',
  numero: '',
  piso: '',
  localidad: '',
  provincia: '',
  cp: '',
  tieneObraSocial: '',
  obraSocial: '',
  tipoContratacion: '',
};

export const PASOS: readonly { numero: PasoPerfil; titulo: string }[] = [
  { numero: 1, titulo: 'Documento' },
  { numero: 2, titulo: 'Datos personales' },
  { numero: 3, titulo: 'Domicilio' },
  { numero: 4, titulo: 'Cobertura y contratación' },
];

const EDAD_MINIMA = 18;

/** Edad cumplida en años enteros. Compara mes y día para no adelantar el cumpleaños. */
function edadEnAnios(nacimiento: Date, hoy: Date): number {
  let anios = hoy.getUTCFullYear() - nacimiento.getUTCFullYear();
  const mes = hoy.getUTCMonth() - nacimiento.getUTCMonth();
  if (mes < 0 || (mes === 0 && hoy.getUTCDate() < nacimiento.getUTCDate())) anios -= 1;
  return anios;
}

export function validarPaso(paso: PasoPerfil, borrador: BorradorPerfil, hoy: Date = new Date()): ErroresPerfil {
  const errores: ErroresPerfil = {};

  if (paso === 1) {
    const numero = borrador.numeroDocumento.trim();
    if (numero === '') {
      errores.numeroDocumento = 'Ingresá tu número de documento';
    } else if (borrador.tipoDocumento === 'DNI') {
      if (!/^\d{7,8}$/.test(numero)) errores.numeroDocumento = 'El DNI son 7 u 8 dígitos, sin puntos';
    } else if (!/^[A-Za-z0-9]{6,}$/.test(numero)) {
      errores.numeroDocumento = 'El pasaporte tiene al menos 6 caracteres';
    }
    return errores;
  }

  if (paso === 2) {
    if (borrador.sexo === '') errores.sexo = 'Elegí una opción';

    const fecha = borrador.fechaNacimiento;
    if (fecha === '') {
      errores.fechaNacimiento = 'Ingresá tu fecha de nacimiento';
    } else {
      const nacimiento = new Date(`${fecha}T00:00:00Z`);
      if (Number.isNaN(nacimiento.getTime())) {
        errores.fechaNacimiento = 'Revisá la fecha';
      } else if (nacimiento.getTime() > hoy.getTime()) {
        errores.fechaNacimiento = 'Revisá la fecha: todavía no llegó';
      } else if (edadEnAnios(nacimiento, hoy) < EDAD_MINIMA) {
        errores.fechaNacimiento = `Tenés que ser mayor de ${EDAD_MINIMA} años`;
      }
    }
    return errores;
  }

  if (paso === 3) {
    if (borrador.calle.trim() === '') errores.calle = 'Ingresá la calle';
    if (borrador.numero.trim() === '') errores.numero = 'Ingresá el número';
    if (borrador.localidad.trim() === '') errores.localidad = 'Ingresá la localidad';
    if (borrador.provincia.trim() === '') errores.provincia = 'Ingresá la provincia';
    if (!/^\d{4}$/.test(borrador.cp.trim())) errores.cp = 'El código postal son 4 dígitos';
    // `piso` es opcional a propósito: hay domicilios sin piso.
    return errores;
  }

  // Paso 4. La obra social NO bloquea: su obligatoriedad está en debate
  // (docs/CONTEXTO.md) y en la maqueta lleva el badge "opcional · a definir".
  if (borrador.tipoContratacion === '') errores.tipoContratacion = 'Elegí el tipo de contratación';
  return errores;
}

export function aPerfil(borrador: BorradorPerfil): PerfilPrevinca {
  if (borrador.sexo === '' || borrador.tipoContratacion === '') {
    throw new Error('El borrador del perfil está incompleto');
  }

  const piso = borrador.piso.trim();

  return {
    tipoDocumento: borrador.tipoDocumento,
    numeroDocumento: borrador.numeroDocumento.trim(),
    sexo: borrador.sexo,
    fechaNacimiento: borrador.fechaNacimiento,
    domicilio: {
      calle: borrador.calle.trim(),
      numero: borrador.numero.trim(),
      ...(piso !== '' ? { piso } : {}),
      localidad: borrador.localidad.trim(),
      provincia: borrador.provincia.trim(),
      cp: borrador.cp.trim(),
    },
    tieneObraSocial: borrador.tieneObraSocial === 'si',
    ...(borrador.tieneObraSocial === 'si' && borrador.obraSocial.trim() !== ''
      ? { obraSocial: borrador.obraSocial.trim() }
      : {}),
    tipoContratacion: borrador.tipoContratacion,
  };
}

/** Para reabrir el wizard desde `/cuenta` con los datos ya cargados. */
export function desdePerfil(perfil: PerfilPrevinca): BorradorPerfil {
  return {
    tipoDocumento: perfil.tipoDocumento,
    numeroDocumento: perfil.numeroDocumento,
    sexo: perfil.sexo,
    fechaNacimiento: perfil.fechaNacimiento,
    calle: perfil.domicilio.calle,
    numero: perfil.domicilio.numero,
    piso: perfil.domicilio.piso ?? '',
    localidad: perfil.domicilio.localidad,
    provincia: perfil.domicilio.provincia,
    cp: perfil.domicilio.cp,
    tieneObraSocial: perfil.tieneObraSocial ? 'si' : 'no',
    obraSocial: perfil.obraSocial ?? '',
    tipoContratacion: perfil.tipoContratacion,
  };
}
```

```tsx
import type { FormEvent } from 'react';
import type { PerfilPrevinca } from '@/mock/session';
import {
  aPerfil,
  BORRADOR_INICIAL,
  PASOS,
  validarPaso,
  type BorradorPerfil,
  type ErroresPerfil,
  type PasoPerfil,
} from '@/features/perfil/validacion';
import { Field } from '@/ui/Field';
import { Input } from '@/ui/Input';
import { Select } from '@/ui/Select';
import { Button } from '@/ui/Button';
import { Badge } from '@/ui/Badge';
import { Card } from '@/ui/Card';

export interface PerfilWizardProps {
  pasoInicial?: PasoPerfil;
  borradorInicial?: BorradorPerfil;
  onGuardar: (perfil: PerfilPrevinca) => void;
}

const TOTAL_PASOS = PASOS.length;

const OPCIONES_DOCUMENTO = [
  { value: 'DNI', label: 'DNI' },
  { value: 'Pasaporte', label: 'Pasaporte' },
];

const OPCIONES_SEXO = [
  { value: 'F', label: 'Femenino' },
  { value: 'M', label: 'Masculino' },
  { value: 'X', label: 'X' },
];

const OPCIONES_OBRA_SOCIAL = [
  { value: 'si', label: 'Sí' },
  { value: 'no', label: 'No' },
];

const OPCIONES_CONTRATACION = [
  { value: 'beneficiario', label: 'Beneficiario' },
  { value: 'tomador', label: 'Tomador' },
  { value: 'tomador-beneficiario', label: 'Tomador y beneficiario' },
];

/** Badge de los campos cuya obligatoriedad Previnca todavía no cerró. */
const EN_DEBATE = <Badge tono="warn">opcional · a definir</Badge>;

export function PerfilWizard({ pasoInicial = 1, borradorInicial, onGuardar }: PerfilWizardProps) {
  const [paso, setPaso] = useState<PasoPerfil>(pasoInicial);
  const [borrador, setBorrador] = useState<BorradorPerfil>(borradorInicial ?? BORRADOR_INICIAL);
  const [errores, setErrores] = useState<ErroresPerfil>({});

  const esUltimo = paso === TOTAL_PASOS;
  const titulo = PASOS.find((p) => p.numero === paso)?.titulo ?? '';

  function actualizar<C extends keyof BorradorPerfil>(campo: C, valor: BorradorPerfil[C]) {
    setBorrador((previo) => ({ ...previo, [campo]: valor }));
  }

  function enviar(evento: FormEvent<HTMLFormElement>) {
    evento.preventDefault();

    const encontrados = validarPaso(paso, borrador);
    if (Object.keys(encontrados).length > 0) {
      setErrores(encontrados);
      return;
    }
    setErrores({});

    if (!esUltimo) {
      setPaso((previo) => (previo + 1) as PasoPerfil);
      return;
    }

    onGuardar(aPerfil(borrador));
  }

  function volver() {
    setErrores({});
    setPaso((previo) => (previo - 1) as PasoPerfil);
  }

  return (
    <Card className="flex flex-col gap-5">
      <div>
        <div className="flex items-baseline justify-between gap-3">
          <span className="text-sm font-medium text-ink-500">{`Paso ${paso} de ${TOTAL_PASOS}`}</span>
          <span className="text-sm text-ink-600">{titulo}</span>
        </div>
        <div
          role="progressbar"
          aria-valuenow={paso}
          aria-valuemin={1}
          aria-valuemax={TOTAL_PASOS}
          aria-label="Progreso del perfil"
          className="mt-2 h-1.5 w-full overflow-hidden rounded-pill bg-ink-100"
        >
          <div
            className="h-full rounded-pill bg-brand transition-[width] duration-300 ease-out-expo"
            style={{ width: `${(paso / TOTAL_PASOS) * 100}%` }}
          />
        </div>

    );
```

```tsx
      <Chip tono="warn" icono={Clock}>
        Pendiente de activación
      </Chip>
    );
  }

  return (
    <Chip tono="danger" icono={CircleAlert}>
      Sin pagar
    </Chip>
  );
}
```

```ts
import { buscarPlan, buscarProducto, type Plan, type Producto } from '@/mock/planes';
import type { Orden, Session } from '@/mock/session';

function pagadasDeTipo(sesion: Session, tipo: Orden['tipo']): Orden[] {
  return sesion.ordenes.filter((orden) => orden.estado === 'pagada' && orden.tipo === tipo);
}

/** La orden pagada más reciente, comparando por fecha ISO (ordenable como texto). */
function masReciente(ordenes: Orden[]): Orden | undefined {
  return ordenes.reduce<Orden | undefined>(
    (mejor, orden) => (mejor === undefined || orden.fecha > mejor.fecha ? orden : mejor),
    undefined,
  );
}

export function planContratado(sesion: Session): Plan | undefined {
  const ultima = masReciente(pagadasDeTipo(sesion, 'plan'));
  return ultima ? buscarPlan(ultima.itemId) : undefined;
}

export function productosContratados(sesion: Session): Producto[] {
  const ids = new Set(pagadasDeTipo(sesion, 'producto').map((orden) => orden.itemId));
  return [...ids].map((id) => buscarProducto(id)).filter((p): p is Producto => p !== undefined);
}

/** ISO `2026-09-01` → `01/09/2026`. A mano, sin `toLocaleDateString`, para que Node y el browser coincidan. */
export function formatearFecha(iso: string): string {
  const [anio, mes, dia] = iso.split('-');
  return `${dia}/${mes}/${anio}`;
}

/** Fecha ficticia de la próxima cuota: un mes después de la última pagada. */
export function proximoVencimiento(sesion: Session): string | undefined {
  const ultima = masReciente(pagadasDeTipo(sesion, 'plan'));
  if (ultima === undefined) return undefined;

  const fecha = new Date(`${ultima.fecha}T00:00:00Z`);
  fecha.setUTCMonth(fecha.getUTCMonth() + 1);

  const anio = fecha.getUTCFullYear().toString();
  const mes = (fecha.getUTCMonth() + 1).toString().padStart(2, '0');
  const dia = fecha.getUTCDate().toString().padStart(2, '0');
  return `${dia}/${mes}/${anio}`;
}
```

```tsx
  { valor: 'todos', label: 'Todos' },
  { valor: 'pagos', label: 'Pagos' },
  { valor: 'contrataciones', label: 'Contrataciones' },
];

/**
 * "Pagos" son las cuotas del plan y "Contrataciones" los servicios on demand:
 * los dos filtros particionan la lista sin dejar movimientos afuera.
 */
export function filtrarMovimientos(ordenes: Orden[], filtro: FiltroMovimientos): Orden[] {
  if (filtro === 'todos') return ordenes;
  if (filtro === 'pagos') return ordenes.filter((orden) => orden.tipo === 'plan');
  return ordenes.filter((orden) => orden.tipo === 'producto');
}

export interface MovimientosListProps {
  ordenes: Orden[];

      <ul className="divide-y divide-ink-100">
        {ordenes.map((orden) => (
          <MovimientoRow key={orden.id} orden={orden} />
        ))}
```

```tsx
          <Seccion titulo="Documento" onEditar={() => setPasoEditado(1)}>
            <p className="text-sm text-ink-900">{`${perfil.tipoDocumento} ${perfil.numeroDocumento}`}</p>
          </Seccion>

          <Seccion titulo="Datos personales" onEditar={() => setPasoEditado(2)}>
            <p className="text-sm text-ink-900">
              {`${ETIQUETA_SEXO[perfil.sexo]} · Nacimiento ${perfil.fechaNacimiento}`}
            </p>
          </Seccion>

          <Seccion titulo="Domicilio" onEditar={() => setPasoEditado(3)}>
            <p className="text-sm text-ink-900">{domicilioLegible(perfil)}</p>
          </Seccion>

          <Seccion titulo="Cobertura y contratación" onEditar={() => setPasoEditado(4)}>
            <div className="flex flex-col gap-2 text-sm text-ink-900">
              <div className="flex flex-wrap items-center gap-2">
                <span>
                  {perfil.tieneObraSocial
                    ? `Obra social: ${perfil.obraSocial ?? 'sin especificar'}`

      ) : (
        <Seccion titulo="Perfil Previnca">
          <p className="text-sm text-ink-600">

          <Icon icono={CreditCard} tono="neutro" tamano="sm" />
          <div className="min-w-0">
            <p className="text-sm font-medium text-ink-900">Visa •••• 4242</p>
            <p className="text-xs text-ink-500">Medio de pago de demostración</p>
          </div>
        </div>
      </Seccion>

      <Seccion titulo="Notificaciones">
        <div className="flex flex-col gap-3">
          <Interruptor id="notif-email" label="Avisos por email" />
          <Interruptor id="notif-push" label="Recordatorio de vencimiento" />
          <p className="text-xs text-ink-500">sin efecto · demo</p>
        </div>
      </Seccion>

      <Seccion titulo="Sesión">
        <a
          href={landingUrl()}
          onClick={() => resetSession()}
          className={clasesBoton({ variante: 'peligro', tamano: 'sm', className: 'self-start' })}
        >
          Cerrar sesión
        </a>
      </Seccion>

      {pasoEditado !== null && perfil ? (
        <Dialog
          abierto
          onCambio={(abierto) => {
            if (!abierto) setPasoEditado(null);
          }}
          titulo="Editar tus datos"
        >
          <PerfilWizard pasoInicial={pasoEditado} borradorInicial={desdePerfil(perfil)} onGuardar={guardar} />
        </Dialog>
      ) : null}
    </div>
  );
}
```
